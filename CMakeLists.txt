cmake_minimum_required(VERSION 3.22)
project(tpersist VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
option(TPERSIST_NAVISERVER "Build NaviServer module entry points" OFF)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(TclStub 8.6 REQUIRED)

# Embed the Tcl adapter so static builds and relocated installs need no source
# tree or external script path. No database client libraries are linked here.
set(TPERSIST_SCRIPT "")
foreach(script library/tdbc.tcl library/schema.tcl)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${script}")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${script}" part HEX)
    string(APPEND TPERSIST_SCRIPT "${part}0a")
endforeach()
configure_file(library/oo.tcl oo.tcl COPYONLY)
configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
string(REGEX REPLACE "(..)" "0x\\1," TPERSIST_SCRIPT "${TPERSIST_SCRIPT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tpersist_script.h"
    "static const char tpersist_script[] = {${TPERSIST_SCRIPT}0};\n")

add_library(tpersist src/library.c src/statements.c src/compiler.c src/expressions.c)
set_target_properties(tpersist PROPERTIES POSITION_INDEPENDENT_CODE ON PREFIX "lib")
target_include_directories(tpersist PRIVATE "${TCL_INCLUDE_PATH}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(tpersist PRIVATE BUILD_tpersist TCL_THREADS TPERSIST_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(tpersist PRIVATE /W4)
else()
    target_compile_options(tpersist PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_SHARED_LIBS)
    target_compile_definitions(tpersist PRIVATE USE_TCL_STUBS)
    target_link_libraries(tpersist PRIVATE "${TCL_STUB_LIBRARY}")
else()
    target_compile_definitions(tpersist PRIVATE STATIC_BUILD)
endif()
if(TPERSIST_NAVISERVER)
    find_path(NAVISERVER_INCLUDE_DIR ns.h REQUIRED)
    target_include_directories(tpersist PRIVATE "${NAVISERVER_INCLUDE_DIR}")
    target_compile_definitions(tpersist PRIVATE USE_NAVISERVER)
endif()
set(PACKAGE_DIR "lib/tpersist${PROJECT_VERSION}")
install(TARGETS tpersist LIBRARY DESTINATION "${PACKAGE_DIR}" ARCHIVE DESTINATION "${PACKAGE_DIR}" RUNTIME DESTINATION "${PACKAGE_DIR}")
install(FILES src/library.h DESTINATION include/tpersist)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl" library/oo.tcl DESTINATION "${PACKAGE_DIR}")
include(CTest)
if(BUILD_TESTING)
    if(BUILD_SHARED_LIBS)
        add_test(NAME tpersist COMMAND "${TCL_TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl" "${CMAKE_CURRENT_BINARY_DIR}")
    else()
        find_library(TCL_RUNTIME_LIBRARY NAMES tcl${TCL_VERSION_MAJOR}.${TCL_VERSION_MINOR} tcl${TCL_VERSION_MAJOR}${TCL_VERSION_MINOR} tcl${TCL_VERSION_MAJOR}${TCL_VERSION_MINOR}t HINTS "${TCL_LIBRARY_DIR}" REQUIRED)
        add_executable(static_smoke tests/static.c)
        target_include_directories(static_smoke PRIVATE "${TCL_INCLUDE_PATH}" src)
        target_link_libraries(static_smoke PRIVATE tpersist "${TCL_RUNTIME_LIBRARY}")
        add_test(NAME static_smoke COMMAND static_smoke)
    endif()
endif()
